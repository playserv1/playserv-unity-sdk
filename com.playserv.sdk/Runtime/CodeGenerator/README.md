# PlayServ Shared DTO Generator

A Unity-friendly, attribute-driven DTO generator.

This system allows you to **declare DTO projections directly in C#** using a `[Shared]` attribute and automatically generates:

- strongly-typed DTOs
- nested DTO hierarchies
- optional mapping code (`From(model)`)
- editor-only, safe code generation

Designed for **Unity projects**, but fully usable in pure C# as well.

using System.Collections.Generic;
using Playserv.Shared;

namespace Playserv.Test
{
    public enum PlayerStatus
    {
        Offline,
        Online,
        Banned
    }

    public sealed class InventoryItem
    {
        public int ItemId { get; set; }
        public string ItemName { get; set; } = "";
        public int Price { get; set; }
    }

    public sealed class Player
    {
        public int Id { get; set; }
        public string Name { get; set; } = "";
        public int Level { get; set; }
        public PlayerStatus Status { get; set; }

        public List<InventoryItem> Inventory { get; set; } = new();
    }

    public sealed class ViewModel
    {
        //ViewModel_Player not exist it autogenerated
        [Shared(typeof(Player), "player", Selection = "{ Id Name Level Inventory }")]
        public ViewModel_Player Player { get; set; }
    }
}
